# TRACE-TIME-PIN plants (cargo xtask plants; docs/plants.md), run on a Skylake-SP host (RAM under /mnt/zram/claude).
# Each command runs the whole build trace (regen.sh) of the plant copy against a RAM copy of the role's trace products
# (target/product/fast/trace of a build of this tree); the tracer under test is the plant copy's. regen.sh traces
# `page_faults mktime` twice, once with the tracer sleeping 2 s at every time read, and fails when the two differ.
# tpin-clock-off: the virtual clock removed (the time entries run their own code: the host's TSC and clocks).
tpin-clock-off | tools/hottext/trace_gdb.py | lit:answer_time(*read) ==> pass | d=$(mktemp -d /mnt/zram/claude/TRACE-TIME-PIN/pt.XXXXXX) && NIJI_HOT_LISTS=$d bash tools/hottext/regen.sh . --products /mnt/zram/claude/TRACE-TIME-PIN/prod | msg:mktime traced with 2 s between time reads differs
# tpin-tsc-host: only the TSC entry is left to the host (the gate in front of the once-a-second check).
tpin-tsc-host | tools/hottext/trace_gdb.py | lit:"niji_machine_rdtsc": "rdtsc", ==>  | d=$(mktemp -d /mnt/zram/claude/TRACE-TIME-PIN/pt.XXXXXX) && NIJI_HOT_LISTS=$d bash tools/hottext/regen.sh . --products /mnt/zram/claude/TRACE-TIME-PIN/prod | msg:mktime traced with 2 s between time reads differs
# tpin-vdso-clock-host: the vDSO's clock_gettime is not recognised (the library's clock read runs the host's clock).
tpin-vdso-clock-host | tools/hottext/trace_gdb.py | lit:in ("clock_gettime", "gettimeofday", "time"): ==> in ("gettimeofday", "time"): | d=$(mktemp -d /mnt/zram/claude/TRACE-TIME-PIN/pt.XXXXXX) && NIJI_HOT_LISTS=$d bash tools/hottext/regen.sh . --products /mnt/zram/claude/TRACE-TIME-PIN/prod | msg:mktime traced with 2 s between time reads differs
