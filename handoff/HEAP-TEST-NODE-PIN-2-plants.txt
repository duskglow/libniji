# HEAP-TEST-NODE-PIN-2 plants (cargo xtask plants; docs/plants.md): the one-node pinning of
# the_heap_keeps_address_space_near_its_use, caught by its assert_on_one_node check.
# A jobrun job is placed in a slot of a few CPUs inside one node, so the test inherits a one-node mask and the pinning
# is a no-op there (HEAP-TEST-NODE-PIN's plants2 survived for that reason). The command therefore widens the inherited
# mask by ONE CPU of another node (the first CPU of the first node that does not hold the slot's first CPU) and prints
# the mask it used; on a one-node host it adds nothing and the plants survive (nothing to catch there).
# pin-skipped: a two-node mask is treated as one node, so nothing is pinned.
# pin-all-cpus: with_cpus sets every CPU instead of the node's.
pin-skipped | crates/core/tests/heap_meta_products.rs | lit:[first, _, ..] => first, ==> [_first, _, ..] => &[], | a=$(taskset -pc $$); a=${a##*: }; f=${a%%[-,]*}; o=; for n in /sys/devices/system/node/node[0-9]*; do [ -e $n/cpu$f ] && continue; o=$(cat $n/cpulist); o=,${o%%[-,]*}; break; done; echo "plant cpus: $a$o"; taskset -c "$a$o" cargo test -p niji --test heap_meta_products the_heap_keeps_address_space_near_its_use -- --nocapture | msg:not one \(pinning skipped or wrong mask\)
pin-all-cpus | crates/core/tests/heap_meta_products.rs | lit:for &c in cpus { ==> for c in 0..8192 { | a=$(taskset -pc $$); a=${a##*: }; f=${a%%[-,]*}; o=; for n in /sys/devices/system/node/node[0-9]*; do [ -e $n/cpu$f ] && continue; o=$(cat $n/cpulist); o=,${o%%[-,]*}; break; done; echo "plant cpus: $a$o"; taskset -c "$a$o" cargo test -p niji --test heap_meta_products the_heap_keeps_address_space_near_its_use -- --nocapture | msg:not one \(pinning skipped or wrong mask\)
